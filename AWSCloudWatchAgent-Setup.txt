###########################################################################################################
##############################################  Commands To use  ##########################################
###########################################################################################################


-----------------------------------------------------------------------------------------------------------
--------------------------------------------- Linux Command Line ------------------------------------------
-----------------------------------------------------------------------------------------------------------


-----------------------------------------------------------------------------------------------------------
|                                      Download and Unpack Client                                         |
-----------------------------------------------------------------------------------------------------------
|                                                                                                         |
| cd /tmp												                                                                          |
| curl -O https://s3.amazonaws.com/amazoncloudwatch-agent/ubuntu/amd64/latest/amazon-cloudwatch-agent.deb |
| sudo dpkg -i -E ./amazon-cloudwatch-agent.deb								                                            |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                        Check If it Installed                                            |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  /opt/aws/amazon-cloudwatch-agent/bin/amazon-cloudwatch-agent-ctl -a status				                      |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                        Start Config Wizard 						                                  |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  sudo /opt/aws/amazon-cloudwatch-agent/bin/amazon-cloudwatch-agent-config-wizard			                  |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                       Edit Config File                                                  |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  sudo nano /opt/aws/amazon-cloudwatch-agent/etc/config.json						                                  |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                       Start Agent With Config                                           |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  sudo /opt/aws/amazon-cloudwatch-agent/bin/amazon-cloudwatch-agent-ctl \				                        |
|    -a fetch-config \											                                                              |
|    -m ec2 \												                                                                      |
|    -c file:/opt/aws/amazon-cloudwatch-agent/etc/config.json \						                                |
|    -s													                                                                          |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                      Enable Agent Start At Launch                                       |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  sudo systemctl enable amazon-cloudwatch-agent						 	                                            |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                         Restart The Service                                             |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  sudo systemctl restart openvpnas									                                                      |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                       Test Logs Manually                                                |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  echo "$(date) CloudWatch agent test line" | sudo tee -a /var/log/openvpnas.log			                    |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




###########################################################################################################
---------------------------------------- Config File Example (Linux) --------------------------------------
###########################################################################################################

{
  "logs": {
    "logs_collected": {
      "files": {
        "collect_list": [
          {
            "file_path": "/var/log/openvpnas.log",
            "log_group_name": "OpenVPN-Access-Logs",
            "log_stream_name": "{instance_id}",
            "timestamp_format": "%Y-%m-%d %H:%M:%S"
          }
        ]
      }
    }
  }
}

-----------------------------------------------------------------------------------------------------------
----------------------------------------  PowerShell (Windows) --------------------------------------------
-----------------------------------------------------------------------------------------------------------


-------------------------------------------------------------------------------------------------------------
|          Download the CloudWatch Agent using the msi installer from the official AWS website              |
-------------------------------------------------------------------------------------------------------------
|													                                                                                  |
| https://docs.aws.amazon.com/AmazonCloudWatch/latest/monitoring/download-cloudwatch-agent-commandline.html |
|													                                                                                  |
-------------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                            Create the Config File                                       |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  notepad C:\Program Files\Amazon\AmazonCloudWatchAgent\config.json					                            |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                              Apply Config File                                          |
-----------------------------------------------------------------------------------------------------------

& "C:\Program Files\Amazon\AmazonCloudWatchAgent\amazon-cloudwatch-agent-ctl.ps1" -a fetch-config -m ec2 -c file:"C:\Program Files\Amazon\AmazonCloudWatchAgent\config.json" -s

-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                         Start the process & restart after config file creation                          |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  & "C:\Program Files\Amazon\AmazonCloudWatchAgent\amazon-cloudwatch-agent-ctl.ps1" `			              |
|     -a fetch-config `											                                                              |
|     -m ec2 `												                                                                    |
|     -c file:"C:\ProgramData\Amazon\AmazonCloudWatchAgent\config.json" `				                          |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|					                                  Start/Check process                                           |
-----------------------------------------------------------------------------------------------------------
|													                                                                                |
|  Start-Service AmazonCloudWatchAgent									                                                  |
|  Get-Service AmazonCloudWatchAgent									                                                    |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                                 Enable Service to autostart on boot                                     |
-----------------------------------------------------------------------------------------------------------
|                                                                                                         |
|  Set-Service -Name AmazonCloudWatchAgent -StartupType Automatic					                                |
|													                                                                                |
-----------------------------------------------------------------------------------------------------------




-----------------------------------------------------------------------------------------------------------
|                           Send a sample EventLog to test if log forwarding is working                   |
-----------------------------------------------------------------------------------------------------------

Write-EventLog -LogName System -Source "EventLog" -EntryType Warning -EventId 1001 -Message "CloudWatch Test - System Log Entry"

-----------------------------------------------------------------------------------------------------------

      "windows_events": {
        "collect_list": [
          {

            "event_levels": ["ERROR", "WARNING", "INFORMATION"],
            "log_group_name": "[Insert-LogGroup-Name]"
          },
          {
            "event_name": "System",
            "event_levels": ["ERROR", "WARNING", "INFORMATION"],
            "log_group_name": "[Insert-LogGroup-Name]"
          }
        ]
      }
    },
    "log_stream_name": "{instance_id}"
  }
}
