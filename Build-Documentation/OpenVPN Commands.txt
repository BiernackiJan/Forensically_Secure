######################################################################################
################################### USER COMMANDS ####################################
######################################################################################


----------------------------------------------
-------- Navigate to script directory --------
----------------------------------------------
|                                            |
|  sudo su                                   |
|  cd /usr/local/openvpn_as/scripts/	     |
|                                            |
----------------------------------------------


--------------------------------------------------------------------------------------
--------------------------- Check users created on system ----------------------------
--------------------------------------------------------------------------------------
|                                                                                    |
|  ./sacli UserPropGet                                                               |
|                                                                                    |
--------------------------------------------------------------------------------------


--------------------------------------------------------------------------------------
--------------------------- Check specific User on system ----------------------------
--------------------------------------------------------------------------------------
|                                                                                    |
|  ./sacli --user <Username> UserPropGet                                             |
|                                                                                    |
--------------------------------------------------------------------------------------


--------------------------------------------------------------------------------------
------------------------- Add a new User for VPN connection --------------------------
--------------------------------------------------------------------------------------
|                                                                                    |
|  ./sacli --user <USER_NAME> --key "type" --value "user_connect" UserPropPut        |
|                                                                                    |
--------------------------------------------------------------------------------------


--------------------------------------------------------------------------------------
------------------------------ Set Password for a User -------------------------------
--------------------------------------------------------------------------------------
|                                                                                    |
|  ./sacli --user '<USER_NAME>' --new_pass '<PASSWORD>' SetLocalPassword             |
|                                                                                    |
--------------------------------------------------------------------------------------





######################################################################################
############################### General Configuration ################################
######################################################################################

--------------------------------------------------------------------------------------
------------------------------ Start/Stop Access Server ------------------------------
--------------------------------------------------------------------------------------
|                                                                                    |
|  ./sacli start                                                                     |
|                                                                                    |
|  ./sacli stop                                                                      |
|                                                                                    |
--------------------------------------------------------------------------------------


--------------------------------------------------------------------------------------
------------------- Change IP in Config if EC2 instance IP changes  ------------------
--------------------------------------------------------------------------------------
|                                                                                    |
|  ./sacli --key "host.name" --value "YOUR_EC2_PUBLIC_IP" ConfigPut                  |
|                                                                                    |
--------------------------------------------------------------------------------------
